main ( )
{void sort(int * * p,int n);

 for (i=0;i<n;i++)

void sort (int * * p,int n)                    /* 排序函数 */
{int i,j,* temp;

{for (j=i+1;j<n;j++)
    {if (* * (p+i)>* * (p+j))            /* 比较后交换整数的地址 */
       {temp=* (p+i); 
        * (p+i)=* (p+j);
        * (p+j)=temp;
        }
     }
 }
